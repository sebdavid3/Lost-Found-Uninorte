Diagrama 3 — Chain of Responsibility

sequenceDiagram
    autonumber
    actor Admin
    participant CC as ClaimsController
    participant PS as PrismaService
    participant IH as IdentityHandler
    participant AH as AvailabilityHandler
    participant EMH as EvidenceMatchHandler
    participant OS as OutboxService

    Admin->>CC: POST /claims/:id/verify (x-user-role: ADMIN)
    CC->>PS: findUnique({ id, include: {user, object, evidences} })
    PS-->>CC: claim
    CC->>CC: if status != PENDING -> BadRequestException
    CC->>IH: new IdentityHandler(prisma)
    CC->>AH: new AvailabilityHandler(prisma)
    CC->>EMH: new EvidenceMatchHandler()
    CC->>IH: setNext(AH)
    IH->>IH: nextHandler = AH
    CC->>AH: setNext(EMH)
    AH->>AH: nextHandler = EMH
    CC->>IH: handle({ claim })

    alt IdentityHandler Success
        IH->>PS: user.findUnique({ id: claim.userId })
        PS-->>IH: user exists
        IH->>AH: super.handle(context)
    else IdentityHandler Failure
        IH->>IH: throw ClaimVerificationException
        Note right of IH: "El usuario con ID {id} no existe
en los registros oficiales."
    end

    alt AvailabilityHandler Success
        AH->>PS: claim.findFirst({ objectId, status: APPROVED, not: userId })
        PS-->>AH: no approvedByOtherUser
        AH->>EMH: super.handle(context)
    else AvailabilityHandler Failure
        AH->>AH: throw ClaimVerificationException
        Note right of AH: "El objeto ya fue reclamado
exitosamente por otra persona."
    end

    alt EvidenceMatchHandler Success
        EMH->>EMH: claim.evidences.some(validTypes.includes(e.type))
        EMH->>EMH: hasValidEvidence == true
        EMH->>EMH: return true
    else EvidenceMatchHandler Failure
        EMH->>EMH: throw ClaimVerificationException
        Note right of EMH: "La reclamación no contiene evidencias
válidas para la categoría {category}.
Se requiere al menos una de: {validTypes}."
    end

    EMH-->>AH: true
    AH-->>IH: true
    IH-->>CC: true
    CC->>CC: $transaction(async tx => { ... })
    CC->>PS: tx.claim.update({ status: APPROVED })
    CC->>OS: enqueueAuditEvent(tx, { action: 'CLAIM_VERIFIED', result: 'SUCCESS' })
    OS-->>CC: outboxEvent created
    PS-->>CC: approvedClaim
    CC-->>Admin: { message: "Reclamación verificada...", claim }

    Note over CC,EMH: Si cualquier handler falla:
CC catch error -> getRejectionDetails()
tx.claim.update({ status: REJECTED })
enqueueAuditEvent(tx, { result: 'FAILURE' })
throw HttpException(CONFLICT)

Diagrama 4 — Outbox Pattern

sequenceDiagram
    autonumber
    actor Client
    participant CS as ClaimsService
    participant PS as PrismaService
    participant OS as OutboxService
    participant OP as OutboxPublisherService
    participant RMQ as RabbitMQ
Queue: audit_events_queue
    participant AC as AuditEventConsumer
    participant ALS as AuditLogService
    participant ALF as AuditLogFactory
    participant PR as PrismaAuditLogRepository
    participant ADB as AuditDB

    Client->>CS: createClaim() / updateClaim() / removeClaim()
    CS->>PS: prisma.$transaction(async tx => { ... })
    PS->>PS: tx.claim.create / update / delete
    CS->>OS: enqueueAuditEvent(tx, AuditEventPayload)
    OS->>PS: tx.outboxEvent.create({
  topic: 'audit.event.created',
  payload: Json,
  status: 'PENDING'
})
    PS-->>OS: OutboxEvent
    PS-->>CS: Transaction Committed
    CS-->>Client: Claim / Object Result

    Note over OP,RMQ: OnModuleInit: setInterval(5000ms)
    OP->>OS: reserveBatch(20)
    OS->>PS: $queryRaw UPDATE "OutboxEvent"
SET status = 'PROCESSING'
WHERE status IN ('PENDING','FAILED')
OR (status='PROCESSING' AND nextAttemptAt < NOW() - 5min)
RETURNING *
    PS-->>OS: events[]
    OS-->>OP: events[]

    loop Cada evento
        OP->>OP: client.emit('audit.event.created', payload)
        OP->>RMQ: Publish audit.event.created
        alt Éxito
            OP->>OS: markPublished(event.id)
            OS->>PS: outboxEvent.update({ status: 'PUBLISHED', publishedAt: now() })
        else Fallo
            OP->>OS: markFailed(id, retryCount, errorMessage)
            OS->>PS: outboxEvent.update({
  status: 'FAILED',
  retryCount+1,
  nextAttemptAt: delay exponencial,
  lastError
})
        end
    end

    RMQ->>AC: Deliver audit.event.created
    AC->>ALS: processAuditEvent(eventData)
    ALS->>ALF: create(eventData, previousHash)
    ALF->>ALF: computeHash(SHA-256) + canonicalizeForHash
    ALF-->>ALS: AuditLogEntryProps
    ALS->>PR: appendWithChain(builder)
    PR->>ADB: prisma.$transaction + tx.auditLog.create({...hash...})
    ADB-->>PR: AuditLog persisted
    PR-->>ALS: OK
    ALS-->>AC: Event processed

Diagrama 5 — Visitor Pattern

classDiagram
    direction TB
    class IVisitor {
        <>
        +visitClaim(claimElement: ClaimElement): void
        +visitEvidence(evidenceElement: EvidenceElement): void
    }

    class IVisitable {
        <>
        +accept(visitor: IVisitor): void
    }

    class ClaimElement {
        +claim: ClaimWithRelations
        +accept(visitor: IVisitor): void
    }

    class EvidenceElement {
        +evidence: Evidence
        +relatedObject: PrismaObject
        +accept(visitor: IVisitor): void
    }

    class AuditVisitor {
        -auditReport: string[]
        +visitClaim(claimElement: ClaimElement): void
        +visitEvidence(evidenceElement: EvidenceElement): void
        +getReport(): string
    }

    class TextSimilarityVisitor {
        -similarityScores: Array~ScoreEntry~
        +visitClaim(_claimElement: ClaimElement): void
        +visitEvidence(evidenceElement: EvidenceElement): void
        +getScores(): Array~ScoreEntry~
        -calculateSimilarity(text1: string, text2: string): number
        -tokenize(text: string): string[]
    }

    IVisitable <|.. ClaimElement
    IVisitable <|.. EvidenceElement
    IVisitor <|.. AuditVisitor
    IVisitor <|.. TextSimilarityVisitor
    ClaimElement --> EvidenceElement : "crea en accept()"
    ClaimsController ..> ClaimElement : "new ClaimElement(claim)"
    ClaimsController ..> AuditVisitor : "new AuditVisitor()"
    ClaimsController ..> TextSimilarityVisitor : "new TextSimilarityVisitor()"
    ClaimsController ..> IVisitable : "claimElement.accept(visitor)"
